#include <stdio.h>
#include <stdlib.h>

int main()
{
    int price_game, membership;

    printf("Enter game price (TL): ");
    scanf("%d", &price_game);

    printf("Enter membership tier (1:Standard, 2:Premium): ");
    scanf("%d", &membership);

    if (membership == 2)
    {
        price_game = (price_game * 80) / 100;
    }

    if (price_game > 1500)
    {
        price_game -= 200;
    }

    printf("Final price to pay: %d TL", price_game);

    return 0;
}
